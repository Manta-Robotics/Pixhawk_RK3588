cmake_minimum_required(VERSION 3.20)
project(manta_control VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(nlohmann_json 3.10 REQUIRED)
add_library(manta_core src/uwb_filter.cpp src/uwb_fusion.cpp)
target_include_directories(manta_core PUBLIC include)
target_link_libraries(manta_core PUBLIC nlohmann_json::nlohmann_json)
target_compile_options(manta_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(manta-replay src/replay.cpp)
target_link_libraries(manta-replay PRIVATE manta_core)
include(CTest)
