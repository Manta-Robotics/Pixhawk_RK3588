cmake_minimum_required(VERSION 3.20)
project(manta_control VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(nlohmann_json 3.10 REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(MAVLINK_GENERATED ${CMAKE_CURRENT_BINARY_DIR}/generated)
add_custom_command(OUTPUT ${MAVLINK_GENERATED}/ardupilotmega/mavlink.h
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../scripts/generate_mavlink.py ${MAVLINK_GENERATED}
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/../scripts/generate_mavlink.py
    COMMENT "Generating official MAVLink C wire protocol" VERBATIM)
add_custom_target(manta_mavlink_headers DEPENDS ${MAVLINK_GENERATED}/ardupilotmega/mavlink.h)
add_library(manta_core src/uwb_filter.cpp src/uwb_fusion.cpp src/uwb_pipeline.cpp src/rover_mapping.cpp)
target_include_directories(manta_core PUBLIC include)
target_link_libraries(manta_core PUBLIC nlohmann_json::nlohmann_json)
target_compile_options(manta_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(manta_follow src/follow_config.cpp src/gps_speed.cpp src/relative_nmpc.cpp src/relative_control.cpp src/follow_preflight.cpp src/follow_controller.cpp src/neural_follow.cpp src/target_estimator.cpp src/raw_recovery.cpp src/intermittent_follow.cpp src/propulsion.cpp src/auto_resume.cpp src/reject_nmpc.cpp src/drift_hold.cpp)
target_link_libraries(manta_follow PUBLIC manta_core)
target_compile_options(manta_follow PRIVATE -Wall -Wextra -Wpedantic)
add_executable(manta-follow-replay src/follow_replay.cpp)
target_link_libraries(manta-follow-replay PRIVATE manta_follow)
add_library(manta_mavlink src/mavlink_core.cpp)
add_dependencies(manta_mavlink manta_mavlink_headers)
target_include_directories(manta_mavlink SYSTEM PUBLIC ${MAVLINK_GENERATED})
target_link_libraries(manta_mavlink PUBLIC manta_core)
target_compile_options(manta_mavlink PRIVATE -Wall -Wextra -Wpedantic)
add_executable(manta-wire-replay src/wire_replay.cpp)
target_link_libraries(manta-wire-replay PRIVATE manta_mavlink)
if(UNIX AND NOT APPLE)
    add_executable(manta-control src/service.cpp src/control_runtime.cpp)
    target_link_libraries(manta-control PRIVATE manta_mavlink manta_follow)
    target_compile_options(manta-control PRIVATE -Wall -Wextra -Wpedantic)
    install(TARGETS manta-control RUNTIME DESTINATION bin)
endif()
add_executable(manta-replay src/replay.cpp)
target_link_libraries(manta-replay PRIVATE manta_core)
include(CTest)
if(BUILD_TESTING)
    add_test(NAME native_uwb_contract
        COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/../scripts/check_native_uwb_contract.py
            $<TARGET_FILE:manta-wire-replay> $<TARGET_FILE:manta-follow-replay>)
    if(TARGET manta-control)
        add_test(NAME native_service
            COMMAND ${Python3_EXECUTABLE}
                ${CMAKE_CURRENT_SOURCE_DIR}/../scripts/check_native_service.py
                $<TARGET_FILE:manta-control>)
    endif()
endif()
